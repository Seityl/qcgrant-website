{{- $Src := .Src }} {{- $Alt := .Alt | default "image" }} {{- $Class := .Class | default "" }} {{- $Width := .Width }} {{- $Height := .Height }} {{- $Loading := .Loading | default "lazy" }} {{- if $Src }} {{- $image := "" }} {{- if or (hasPrefix $Src "http://") (hasPrefix $Src "https://") }} {{/* External image */}} {{- else }} {{- $image = resources.Get $Src }} {{- if $image }} {{/* Process image with Hugo */}} {{- $img := $image }} {{- if and $Width $Height }} {{- $img = $image.Resize (printf "%dx%d webp" $Width $Height) }} {{- else if $Width }} {{- $img = $image.Resize (printf "%dx webp" $Width) }} {{- end }} {{- else }} {{/* Fallback for static images */}} {{- end }} {{- end }} {{- end }}
